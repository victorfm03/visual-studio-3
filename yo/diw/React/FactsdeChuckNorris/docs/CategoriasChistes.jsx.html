import { useEffect, useState } from "react";

/**
 * @fileoverview Componente CrearCategoria para gestionar categorías de chistes
 * @description Este componente es responsable de obtener las categorías disponibles
 * desde la API pública de Chuck Norris y presentarlas en un selector (dropdown)
 * para que el usuario pueda elegir una categoría
 */

/**
 * Componente CrearCategoria
 * 
 * Este componente se encarga de:
 * 1. Obtener las categorías de chistes disponibles desde la API de Chuck Norris
 * 2. Mostrar un select (dropdown) con todas las categorías disponibles
 * 3. Manejar la selección del usuario y notificar al componente padre
 * 4. Gestionar el estado de carga de datos desde la API
 * 
 * @component
 * @param {Object} props - Propiedades del componente
 * @param {Function} props.handler - Función callback que se ejecuta cuando se selecciona una categoría
 *                                   Recibe como parámetro el nombre de la categoría seleccionada
 * 
 * @returns {JSX.Element} Un fragmento que contiene el selector de categorías
 * 
 * @example
 * // Uso del componente
 * <CrearCategoria handler={(categoria) => console.log("Seleccionada:", categoria)} />
 */
function CrearCategoria({handler}) {
  // Estado que indica si la carga de categorías desde la API está en progreso
  // Inicialmente true para disparar la obtención de datos en el useEffect
  const [isLoading, setIsLoading] = useState(true);
  
  // Estado que almacena el array de nombres de categorías obtenidas de la API
  // Ejemplo: ["animal", "career", "celebrity", "dev", ...]
  const [categorias, setCategories] = useState([]);
  
  // Estado que almacena la categoría actualmente seleccionada en el select
  // Comienza vacío y se establece con la primera categoría al cargar los datos
  const [selectedCategory, setSelectedCategory] = useState("");

  /**
   * Hook useEffect para obtener las categorías de la API al montar el componente
   * Solo se ejecuta cuando el estado isLoading es true
   */
  useEffect(() => {
    /**
     * Función asincrónica que obtiene las categorías desde la API de Chuck Norris
     * URL: https://api.chucknorris.io/jokes/categories
     * Retorna un array con los nombres de todas las categorías disponibles
     */
    async function fetchCategories() {
      try {
        // Realiza la solicitud GET a la API de Chuck Norris
        let res = await fetch("https://api.chucknorris.io/jokes/categories");
        
        // Convierte la respuesta de la API a un array JSON
        let array_category = await res.json();
        
        // Actualiza el estado con las categorías obtenidas
        setCategories(array_category);
        
        // Indica que la carga ha finalizado
        setIsLoading(false);

        // Si hay al menos una categoría disponible
        if (array_category.length > 0) {
          // Establece la primera categoría como seleccionada por defecto
          setSelectedCategory(array_category[0]);
          
          // Notifica al componente padre sobre la categoría seleccionada
          handler(array_category[0]);
        }
      } catch (e) {
        // Manejo de errores: muestra un alert si la API no responde
        alert("Error: ", e);
      }
    }

    // Solo ejecuta la función si está en estado de carga
    if (isLoading) {
      fetchCategories();
    }
  }, [isLoading]); // Dependencia: se ejecuta cuando isLoading cambia

  return (
    <>
      {/* Selector (dropdown) de categorías */}
      <select
        name="categorias"
        // El valor actual del select es la categoría seleccionada
        value={selectedCategory}
        // Maneja el cambio cuando el usuario selecciona una nueva categoría
        onChange={(e) => {
          // Obtiene el valor seleccionado del evento
          const value = e.target.value;
          
          // Actualiza el estado local con la nueva categoría
          setSelectedCategory(value);
          
          // Notifica al componente padre sobre la selección
          handler(value);
        }}
        className="form-select" // Clase Bootstrap para estilos
      >
        {/* Mapea las categorías para crear opciones */}
        {categorias.map((item) => (
          // Crea un elemento option para cada categoría del array
          // key y value utilizan el nombre de la categoría (asume que son únicos)
          <option key={item} value={item}>
            {item}
          </option>
        ))}
      </select>
    </>
  );
}

export default CrearCategoria;
