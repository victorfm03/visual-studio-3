/**
 * @fileoverview Componente principal de la aplicación FactsdeChuckNorris
 * @description Este archivo contiene el componente raíz de la aplicación que gestiona
 * la selección de categorías de chistes de Chuck Norris mediante una interfaz de usuario
 * construida con React y Bootstrap.
 */

import { useState } from "react";
import CrearCategoria from './CategoriasChistes';
import "bootstrap/dist/css/bootstrap.min.css";
import "bootstrap/dist/js/bootstrap.bundle.min.js";

/**
 * Componente App - Componente raíz de la aplicación
 * 
 * Responsabilidades principales:
 * - Gestionar el estado de la categoría seleccionada por el usuario
 * - Renderizar el título principal de la aplicación
 * - Renderizar el componente hijo CrearCategoria para permitir la selección
 * - Pasar la función handleCategoria como prop al componente hijo para comunicación
 * 
 * @component
 * @returns {React.ReactElement} Elemento JSX que contiene la estructura principal de la aplicación
 * 
 * @example
 * // Uso del componente
 * <App />
 */
function App(){
    // Estado para almacenar la categoría de chiste actualmente seleccionada
    // Inicialmente vacío hasta que el usuario seleccione una categoría
    const [categoriaSeleccionada, setCategoriaSeleccionada] = useState("");
    
    /**
     * Función handler que se ejecuta cuando cambia la selección de categoría
     * Actualiza el estado local y registra la selección en la consola para debugging
     * 
     * @param {string} categoria - El nombre de la categoría seleccionada por el usuario
     * @returns {void}
     */
    const handleCategoria =(categoria)=> {
        // Actualiza el estado con la nueva categoría seleccionada
        setCategoriaSeleccionada(categoria);
        // Registra la categoría en la consola para propósitos de depuración y desarrollo
        console.log("Categoria seleccionada:",categoria);
    };

    return (
        // Contenedor principal con espaciado top de Bootstrap (mt-4)
        <div className="container mt-4">
            {/* Título principal de la aplicación */}
            <h1 className="mb-3">Chuck Norris Facts</h1>
            
            {/* Componente que gestiona la selección de categorías */}
            {/* Pasa la función handleCategoria como prop para que el componente hijo pueda notificar cambios */}
            <CrearCategoria handler={handleCategoria} />
        </div>
    );

}

export default App
